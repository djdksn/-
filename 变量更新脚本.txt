import { registerSchema } from './sillytavern/variable-schema.js';

// ================================================================
// 1. Schema 注册
// ================================================================

export function registerAllSchemas() {
  const schemas = buildSchemas();
  for (const [path, rule] of Object.entries(schemas)) {
    registerSchema({ path, ...rule });
  }
  console.log('[变量脚本] Registered %d schema rules', Object.keys(schemas).length);
}

function buildSchemas() {
  const rules = {};

  // ── User 信息 (数值型) ──────────────────────────────────────
  const userNumberFields = [
    'User信息.性交总次数',
    'User信息.接受性交次数',
    'User信息.接受口交次数',
    'User信息.接受肛交次数',
    'User信息.接受色情按摩次数',
    'User信息.发生性关系总人数',
    'User信息.年龄',
  ];
  for (const f of userNumberFields) {
    rules[f] = { type: 'number', min: 0 };
  }

  // ── User 信息 (字符型) ──────────────────────────────────────
  rules['User信息.姓名'] = { type: 'string' };
  rules['User信息.所属部门'] = { type: 'string' };
  rules['User信息.职务'] = { type: 'string' };

  // ── NPC 花名册 (通配符) ────────────────────────────────────
  const npc = 'NPC花名册.*';

  // 好感度枚举
  rules[`${npc}.动态数据.人物好感度`] = {
    type: 'string',
    enum: ['厌恶', '一般', '友善', '爱慕'],
  };

  // NPC 经历 — 数值字段
  const npcExpNumberFields = [
    '性行为次数', '性交次数', '口交次数',
    '肛交次数', '足交次数',
  ];
  for (const f of npcExpNumberFields) {
    rules[`${npc}.动态数据.经历.${f}`] = { type: 'number', min: 0 };
  }

  // NPC 社会情况 — 数值字段
  rules[`${npc}.静态数据.社会情况.年龄`] = { type: 'number', min: 1, max: 120 };

  // NPC 身体状态 — 即时痕迹 (字符串)
  const bodyParts = ['口', '腋', '胸', '腹', '下体', '臀', '腿', '足'];
  for (const part of bodyParts) {
    rules[`${npc}.动态数据.身体状态.即时痕迹.${part}`] = { type: 'string' };
  }
  rules[`${npc}.动态数据.身体状态.整体反馈`] = { type: 'string' };

  // NPC 穿着字段
  const clothingFields = ['上衣', '下衣', '鞋', '袜', '配饰'];
  for (const f of clothingFields) {
    rules[`${npc}.动态数据.人物穿着.${f}`] = { type: 'string' };
  }

  // NPC 内心想法
  rules[`${npc}.动态数据.内心想法`] = { type: 'string' };

  // ── 课程 (通配符) ──────────────────────────────────────────
  const course = '课程.*';
  rules[`${course}.动态数据.课程进展`] = {
    type: 'string',
    enum: ['未开始', '进行中', '已完成'],
  };

  // ── 当前时间 ──────────────────────────────────────────────
  rules['当前时间.日期'] = { type: 'string' };
  rules['当前时间.时间'] = { type: 'string' };
  rules['当前地点'] = { type: 'string' };

  return rules;
}

/**
 * 将初始变量写入指定 chat。
 * @param {import('./sillytavern-store.js').SillytavernStore} store
 * @param {string} chatId
 */
export async function initChatVariables(store, chatId) {
  const chat = store.chats.find(c => c.id === chatId);
  if (!chat) {
    console.error('[变量脚本] Chat not found:', chatId);
    return;
  }

  const initial = getInitialVariables();

  // 深度合并：保留用户可能已经修改过的字段
  const merged = deepMerge(initial, chat.variables || {});

  // 更新 chat 的 variables
  const updated = { ...chat, variables: merged, updatedAt: Date.now() };
  await store._db.table('chats').put(updated);
  store.chats = store.chats.map(c => c.id === updated.id ? updated : c);
  store._notify();

  console.log('[变量脚本] Chat variables initialized for', chatId);
}

function deepMerge(base, override) {
  const result = { ...base };
  for (const key of Object.keys(override)) {
    if (override[key] && typeof override[key] === 'object' && !Array.isArray(override[key]) && base[key] && typeof base[key] === 'object') {
      result[key] = deepMerge(base[key], override[key]);
    } else {
      result[key] = override[key];
    }
  }
  return result;
}
